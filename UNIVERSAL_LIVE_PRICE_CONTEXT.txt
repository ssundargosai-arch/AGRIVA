def get_farmer_crop(
    crop_id,
    current_user: User,
    db: Session,
) -> Crop:
    crop = (
        db.query(Crop)
        .join(
            FarmField,
            Crop.field_id
            == FarmField.id,
        )
        .join(
            Farm,
            FarmField.farm_id
            == Farm.id,
        )
        .join(
            FarmerProfile,
            Farm.farmer_id
            == FarmerProfile.id,
        )
        .filter(
            Crop.id == crop_id,
            FarmerProfile.user_id
            == current_user.id,
        )
        .first()
    )

    if crop is None:
        raise HTTPException(
            status_code=(
                status.HTTP_404_NOT_FOUND
            ),
            detail=(
                "Crop not found or does not "
                "belong to the logged-in farmer"
            ),
        )

    return crop


def get_latest_quality(
    crop_id,
    db: Session,
) -> CropQualityPrediction | None:
    rows = (
        db.query(
            CropQualityPrediction
        )
        .filter(
            CropQualityPrediction.crop_id
            == crop_id
        )
        .all()
    )

    if not rows:
        return None

    return rows[-1]


@router.post(
    "/price-prediction",
    response_model=(
        PricePredictionResponse
    ),
)
def create_price_prediction(
    payload: PricePredictionRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        require_farmer
    ),
):
    crop: Crop | None = None

    crop_name: str | None = None
    quality: str | None = None

    # =============================================
    # REGISTERED CROP MODE
    # =============================================

    if payload.crop_id is not None:
        crop = get_farmer_crop(
            crop_id=payload.crop_id,
            current_user=current_user,
            db=db,
        )

        crop_name = crop.name

        latest_quality = (
            get_latest_quality(
                crop_id=crop.id,
                db=db,
            )
        )

        if latest_quality is not None:
            quality = (
                latest_quality.grade
            )

        elif payload.quality is not None:
            quality = payload.quality

        else:
            raise HTTPException(
                status_code=404,
                detail=(
                    "No crop quality prediction "
                    "found. Run AGRIVA Crop "
                    "Quality analysis first or "
                    "provide quality manually."
                ),
            )

    # =============================================
    # MANUAL / LEGACY MODE
    # =============================================

    else:
        if not payload.crop:
            raise HTTPException(
                status_code=422,
                detail=(
                    "Either crop_id or crop "
                    "must be provided."
                ),
            )

        if payload.quality is None:
            raise HTTPException(
                status_code=422,
                detail=(
                    "Quality is required when "
                    "crop_id is not provided."
                ),
            )

        crop_name = (
            payload.crop.strip()
        )

        quality = (
            payload.quality
        )

    if crop_name is None:
        raise HTTPException(
            status_code=422,
            detail="Crop name could not be resolved.",
        )

    if quality is None:
        raise HTTPException(
            status_code=422,
            detail="Crop quality could not be resolved.",
        )

    # =============================================
    # PRICE INTELLIGENCE CALCULATION
    # =============================================

    try:
        result = (
            calculate_price_prediction(
                crop=crop_name,
                location=payload.location,
                quality=quality,
                quantity=payload.quantity,
                season=payload.season,
                historical_prices=(
                    payload.historical_prices
                ),
            )
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=422,
            detail=str(exc),
        ) from exc

    # =============================================
    # STORE EXISTING CORE PREDICTION
    #
    # No DB migration is required in this step.
    # Rich live market metadata is returned by
    # the API but not persisted yet.
    # =============================================

    # Phase 6C price-unit safety.
    # Service response prices follow result["unit"].
    # Database *_per_kg columns must always store INR/kg.
    try:
        phase6c_current_price_per_kg = price_to_per_kg(
            result["current_price"],
            result.get("unit"),
        )

        phase6c_predicted_price_per_kg = price_to_per_kg(
            result["predicted_price"],
            result.get("unit"),
        )

        phase6c_expected_min_per_kg = price_to_per_kg(
            result["expected_range_min"],
            result.get("unit"),
        )

        phase6c_expected_max_per_kg = price_to_per_kg(
            result["expected_range_max"],
            result.get("unit"),
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=422,
            detail=str(exc),
        ) from exc

    prediction = PricePrediction(
        user_id=current_user.id,

        crop_id=(
            crop.id
            if crop is not None
            else None
        ),

        crop_name=crop_name,

        market_name=(
            payload.location.strip()
        ),

        quality=quality,

        quantity=payload.quantity,

        season=(
            payload.season.strip()
        ),

        historical_prices=result[
            "historical_prices"
        ],

        current_price_per_kg=(
            phase6c_current_price_per_kg
        ),

        predicted_price_per_kg=(
            phase6c_predicted_price_per_kg
        ),

        expected_min_price=(
            phase6c_expected_min_per_kg
        ),

        expected_max_price=(
            phase6c_expected_max_per_kg
        ),

        confidence=result[
            "confidence"
        ],

        trend=result[
            "trend"
        ],
    )

    try:
        db.add(prediction)
        db.commit()
        db.refresh(prediction)

    except Exception as exc:
        db.rollback()

        raise HTTPException(
            status_code=500,
            detail=(
                "Unable to store price "
                "prediction."
            ),
        ) from exc

    # =============================================
    # CHART DATA
    # =============================================

    chart_data: list[
        PriceHistoryPoint
    ] = []

    for index, price in enumerate(
        result["historical_prices"],
        start=1,
    ):
        chart_data.append(
            PriceHistoryPoint(
                label=f"History {index}",
                price=float(price),
            )
        )

    chart_data.append(
        PriceHistoryPoint(
            label="Current",
            price=float(
                result[
                    "current_price"
                ]
            ),
        )
    )

    chart_data.append(
        PriceHistoryPoint(
            label="Predicted",
            price=float(
                result[
                    "predicted_price"
                ]
            ),
        )
    )

    # =============================================
    # PRICE INTELLIGENCE RESPONSE
    # =============================================

    return PricePredictionResponse(
        id=str(prediction.id),

        crop_id=(
            str(prediction.crop_id)
            if prediction.crop_id
            else None
        ),

        crop=prediction.crop_name,

        location=(
            prediction.market_name
        ),

        quality=(
            prediction.quality
            or quality
        ),

        quantity=(
            prediction.quantity
            if prediction.quantity
            is not None
            else payload.quantity
        ),

        season=(
            prediction.season
            or payload.season
        ),

                current_price=result["current_price"],

                predicted_price=result["predicted_price"],

                expected_range_min=result["expected_range_min"],

                expected_range_max=result["expected_range_max"],

        confidence=(
            prediction.confidence
            if prediction.confidence
            is not None
            else result[
                "confidence"
            ]
        ),

        trend=(
            prediction.trend
            or result[
                "trend"
            ]
        ),

        historical_prices=(
            prediction.historical_prices
            or []
        ),

        chart_data=chart_data,

        demo_mode=result[
            "demo_mode"
        ],

        data_mode=result[
            "data_mode"
        ],

        unit=result[
            "unit"
        ],

        market_data=result[
            "market_data"
        ],

        prediction_factors=result[
            "prediction_factors"
        ],

        price_change_percentage=result[
            "price_change_percentage"
        ],

        disclaimer=DISCLAIMER,
    )

_CACHE_LOCK = Lock()


def _env_flag(
    name: str,
    default: str,
) -> bool:
    return (
        os.getenv(
            name,
            default,
        )
        .strip()
        .lower()
        in {
            "1",
            "true",
            "yes",
            "on",
        }
    )


def _env_float(
    name: str,
    default: float,
    *,
    minimum: float,
) -> float:
    raw = os.getenv(
        name,
        str(default),
    ).strip()

    try:
        value = float(raw)

    except ValueError:
        return default

    return max(
        value,
        minimum,
    )


def _env_int(
    name: str,
    default: int,
    *,
    minimum: int,
    maximum: int,
) -> int:
    raw = os.getenv(
        name,
        str(default),
    ).strip()

    try:
        value = int(raw)

    except ValueError:
        return default

    return max(
        minimum,
        min(
            value,
            maximum,
        ),
    )


def _cache_path() -> Path:
    configured = os.getenv(
        "MANDI_CACHE_FILE",
        "",
    ).strip()

    if configured:
        return Path(configured)

    return (
        BACKEND_ROOT
        / ".cache"
        / "mandi_price_cache.json"
    )


def _cache_max_age() -> float:
    # Default: 3 days
    return _env_float(
        "MANDI_CACHE_MAX_AGE_SECONDS",
        259200.0,
        minimum=60.0,
    )


def _cache_enabled() -> bool:
    return _env_flag(
        "MANDI_CACHE_ENABLED",
        "true",
    )


def _cache_key(
    *,
    commodity: str,
    location: str,
    state: str | None,
) -> str:
    values = [
        commodity.strip().casefold(),
        location.strip().casefold(),
        (
            state.strip().casefold()
            if state
            else ""
        ),
    ]

    return "|".join(values)


def _read_cache_payload() -> dict[str, Any]:
    path = _cache_path()

    if not path.exists():
        return {
            "version": 1,
            "entries": {},
        }

    try:
        with path.open(
            "r",
            encoding="utf-8",
        ) as file:
            payload = json.load(file)

    except (
        OSError,
        ValueError,
        TypeError,
    ):
        return {
            "version": 1,
            "entries": {},
        }

    if not isinstance(
        payload,
        dict,
    ):
        return {
            "version": 1,
            "entries": {},
        }

    entries = payload.get(
        "entries"
    )

    if not isinstance(
        entries,
        dict,
    ):
        payload["entries"] = {}

    return payload


def _save_cached_result(
    *,
    commodity: str,
    location: str,
    state: str | None,
    result: MandiPriceResult,
) -> None:
    if not _cache_enabled():
        return

    key = _cache_key(
        commodity=commodity,
        location=location,
        state=state,
    )

    path = _cache_path()

    try:
        path.parent.mkdir(
            parents=True,
            exist_ok=True,
        )

        with _CACHE_LOCK:
            payload = (
                _read_cache_payload()
            )

            entries = payload.setdefault(
                "entries",
                {},
            )

            entries[key] = {
                "saved_at": time.time(),
                "result": asdict(
                    result
                ),
            }

            temporary = (
                path.with_suffix(
                    path.suffix
                    + ".tmp"
                )
            )

            with temporary.open(
                "w",
                encoding="utf-8",
            ) as file:
                json.dump(
                    payload,
                    file,
                    ensure_ascii=False,
                    indent=2,
                )

            os.replace(
                temporary,
                path,
            )

    except OSError:
        # Cache failure must never break
        # the main price prediction.
        return


def _load_cached_result(
    *,
    commodity: str,
    location: str,
    state: str | None,
) -> MandiPriceResult | None:
    if not _cache_enabled():
        return None

    key = _cache_key(
        commodity=commodity,
        location=location,
        state=state,
    )

    payload = _read_cache_payload()

    entries = payload.get(
        "entries",
        {},
    )

    if not isinstance(
        entries,
        dict,
    ):
        return None

    entry = entries.get(key)

    if not isinstance(
        entry,
        dict,
    ):
        return None

    saved_at_raw = entry.get(
        "saved_at"
    )

    try:
        saved_at = float(
            saved_at_raw
        )

    except (
        TypeError,
        ValueError,
    ):
        return None

    age = (
        time.time()
        - saved_at
    )

    if age < 0:
        age = 0

    if age > _cache_max_age():
        return None

    raw_result = entry.get(
        "result"
    )

    if not isinstance(
        raw_result,
        dict,
    ):
        return None

    modal_price = parse_price(
        raw_result.get(
            "modal_price"
        )
    )

    if modal_price is None:
        return None

    cached_at = (
        datetime.fromtimestamp(
            saved_at,
            tz=timezone.utc,
        )
        .isoformat()
    )

    return MandiPriceResult(
        commodity=str(
            raw_result.get(
                "commodity",
                commodity,
            )
        ),

        state=_optional_text(
            raw_result.get(
                "state"
            )
        ),

        district=_optional_text(
            raw_result.get(
                "district"
            )
        ),

        market=_optional_text(
            raw_result.get(
                "market"
            )
        ),

        variety=_optional_text(
            raw_result.get(
                "variety"
            )
        ),

        grade=_optional_text(
            raw_result.get(
                "grade"
            )
        ),

        arrival_date=(
            _optional_text(
                raw_result.get(
                    "arrival_date"
                )
            )
        ),

        min_price=parse_price(
            raw_result.get(
                "min_price"
            )
        ),

        max_price=parse_price(
            raw_result.get(
                "max_price"
            )
        ),

        modal_price=modal_price,

        source=str(
            raw_result.get(
                "source",
                (
                    "AGMARKNET / "
                    "data.gov.in"
                ),
            )
        ),

        unit=str(
            raw_result.get(
                "unit",
                "INR/quintal",
            )
        ),

        data_mode="CACHED",

        match_level=str(
            raw_result.get(
                "match_level",
                "DISTRICT",
            )
        ),

        cached_at=cached_at,
    )

def normalize_commodity(
    crop: str,
) -> str:
    cleaned = crop.strip()

    if not cleaned:
        raise MandiPriceError(
            "Crop name is required."
        )

    return COMMODITY_NAMES.get(
        cleaned.casefold(),
        cleaned.title(),
    )


def normalize_location(
    location: str,
) -> str:
    cleaned = " ".join(
        location.strip().split()
    )

    if not cleaned:
        raise MandiPriceError(
            "Market or location is required."
        )

    return cleaned.title()

def resolve_maharashtra_location(
    location: str,
) -> tuple[str | None, str | None]:
    normalized = " ".join(
        location.strip().split()
    )

    key = normalized.casefold()

    if key in MAHARASHTRA_DISTRICTS:
        return (
            normalized.title(),
            "Maharashtra",
        )

    district = MAHARASHTRA_MARKET_DISTRICT.get(
        key
    )

    if district:
        return (
            district,
            "Maharashtra",
        )

    return (
        None,
        None,
    )


def split_location_state(
    location: str,
) -> tuple[str, str | None]:
    raw_parts = [
        part.strip()
        for part in location.split(",")
        if part.strip()
    ]

    if not raw_parts:
        raise MandiPriceError(
            "Market or location is required."
        )

    location_name = normalize_location(
        raw_parts[0]
    )

    # Explicit state always wins.
    if len(raw_parts) >= 2:
        explicit_state = " ".join(
            raw_parts[1:]
        ).title()

        return (
            location_name,
            explicit_state,
        )

    # Maharashtra-focused AGRIVA resolver.
    _, detected_state = (
        resolve_maharashtra_location(
            location_name
        )
    )

    return (
        location_name,
        detected_state,
    )


def _optional_text(
    value: Any,
) -> str | None:
    if value is None:
        return None

    cleaned = str(
        value
    ).strip()

    return (
        cleaned
        if cleaned
        else None
    )


# =========================================================
# PARSING
# =========================================================

def parse_price(
    value: Any,
) -> float | None:
    if value is None:
        return None

    text = str(
        value
    ).strip()

    if not text:
        return None

    try:
        number = float(text)

    except (
        TypeError,
        ValueError,
    ):
        return None

    if number <= 0:
        return None

    return number


def parse_date(
    value: str | None,
) -> datetime:
    if not value:
        return datetime.min

    value = value.strip()

    formats = (
        "%d/%m/%Y",
        "%d-%m-%Y",
        "%Y-%m-%d",
        "%Y/%m/%d",
    )

    for date_format in formats:
        try:
            return datetime.strptime(
                value,
                date_format,
            )

        except ValueError:
            continue

    return datetime.min


# =========================================================
# RECORD MATCHING
# =========================================================

def record_matches_location(
    record: dict[str, Any],
    location: str,
) -> bool:
    expected = (
        location
        .strip()
        .casefold()
    )

    market = str(
        record.get(
            "market",
            "",
        )
    ).strip().casefold()

    district = str(
        record.get(
            "district",
            "",
        )
    ).strip().casefold()

    return (
        expected == market
        or expected == district
        or expected in market
        or expected in district
    )


def choose_latest_record(
    records: list[
        dict[str, Any]
    ],
    location: str,
) -> dict[str, Any] | None:
    if not records:
        return None

    matching_records = [
        record
        for record in records
        if record_matches_location(
            record,
            location,
        )
    ]

    valid_records = [
        record
        for record in matching_records
        if parse_price(
            record.get(
                "modal_price"
            )
        )
        is not None
    ]

    if not valid_records:
        return None

    return max(
        valid_records,
        key=lambda record: (
            parse_date(
                _optional_text(
                    record.get(
                        "arrival_date"
                    )
                )
            )
        ),
    )


# =========================================================
# GOVERNMENT API
# =========================================================

def request_records(
    *,
    api_key: str,
    commodity: str | None,
    location: str,
    location_field: str,
    limit: int = 50,
) -> list[dict[str, Any]]:

    params = {
        "api-key": api_key,
        "format": "json",
        "offset": 0,
        "limit": limit,
    }

    if commodity:
        params["filters[commodity]"] = (
            commodity
        )

    params[
        f"filters[{location_field}]"
    ] = location

    timeout = httpx.Timeout(
        connect=_env_float(
            "MANDI_API_CONNECT_TIMEOUT",
            4.0,
            minimum=1.0,
        ),

        read=_env_float(
            "MANDI_API_READ_TIMEOUT",
            6.0,
            minimum=2.0,
        ),

        write=5.0,
        pool=5.0,
    )

    attempts = _env_int(
        "MANDI_API_MAX_ATTEMPTS",
        2,
        minimum=1,
        maximum=3,
    )

    last_exception: (
        Exception | None
    ) = None

    with httpx.Client(
        timeout=timeout,
        follow_redirects=True,
        headers={
            "Accept":
                "application/json",

            "User-Agent":
                "AGRIVA/1.0",
        },
    ) as client:

        for attempt in range(
            attempts
        ):
            try:
                response = client.get(
                    DATA_GOV_API_URL,
                    params=params,
                )

                response.raise_for_status()

                try:
                    payload = (
                        response.json()
                    )

                except ValueError as exc:
                    raise MandiPriceError(
                        "Government mandi-price "
                        "service returned invalid "
                        "JSON data."
                    ) from exc

                if not isinstance(
                    payload,
                    dict,
                ):
                    raise MandiPriceError(
                        "Government mandi-price "
                        "service returned an "
                        "unexpected response."
                    )

                records = payload.get(
                    "records",
                    [],
                )

                if not isinstance(
                    records,
                    list,
                ):
                    return []

                return [
                    record
                    for record
                    in records
                    if isinstance(
                        record,
                        dict,
                    )
                ]

            except (
                httpx.TimeoutException
            ) as exc:
                last_exception = exc

            except (
                httpx.HTTPStatusError
            ) as exc:
                status_code = (
                    exc
                    .response
                    .status_code
                )

                if status_code in {
                    401,
                    403,
                }:
                    raise MandiAuthError(
                        "Government mandi API "
                        "rejected the API key."
                    ) from exc

                if (
                    status_code == 429
                    or status_code >= 500
                ):
                    last_exception = exc

                else:
                    raise MandiPriceError(
                        "Government mandi-price "
                        "service returned HTTP "
                        f"{status_code}."
                    ) from exc

            except (
                httpx.RequestError
            ) as exc:
                last_exception = exc

            # Short controlled backoff.
            if (
                attempt
                < attempts - 1
            ):
                time.sleep(
                    0.5
                    * (
                        2 ** attempt
                    )
                )

    raise (
        MandiServiceUnavailableError(
            "Government mandi-price "
            "service is temporarily "
            "unavailable."
        )
    ) from last_exception


def _resolve_location_record(
    *,
    api_key: str,
    location: str,
) -> dict[str, Any] | None:
    """
    Resolve a farmer-entered district or
    market using Government mandi data.

    Commodity is intentionally omitted
    so location detection does not fail
    just because the selected crop has
    no current record at that location.
    """

    for location_field in (
        "district",
        "market",
    ):
        records = request_records(
            api_key=api_key,
            commodity=None,
            location=location,
            location_field=location_field,
            limit=100,
        )

        record = choose_latest_record(
            records,
            location,
        )

        if record is not None:
            return record

    return None

def _state_market_aggregate(
    *,
    records: list[
        dict[str, Any]
    ],
    commodity: str,
    state: str,
) -> MandiPriceResult | None:
    valid_records = [
        record
        for record in records
        if parse_price(
            record.get(
                "modal_price"
            )
        )
        is not None
    ]

    if not valid_records:
        return None

    latest_date = max(
        parse_date(
            _optional_text(
                record.get(
                    "arrival_date"
                )
            )
        )
        for record
        in valid_records
    )

    latest_records = [
        record
        for record
        in valid_records
        if parse_date(
            _optional_text(
                record.get(
                    "arrival_date"
                )
            )
        )
        == latest_date
    ]

    if not latest_records:
        latest_records = (
            valid_records
        )

    modal_prices = [
        price
        for record
        in latest_records
        if (
            price := parse_price(
                record.get(
                    "modal_price"
                )
            )
        )
        is not None
    ]

    if not modal_prices:
        return None

    min_prices = [
        price
        for record
        in latest_records
        if (
            price := parse_price(
                record.get(
                    "min_price"
                )
            )
        )
        is not None
    ]

    max_prices = [
        price
        for record
        in latest_records
        if (
            price := parse_price(
                record.get(
                    "max_price"
                )
            )
        )
        is not None
    ]

    arrival_date = (
        _optional_text(
            latest_records[0].get(
                "arrival_date"
            )
        )
    )

    return MandiPriceResult(
        commodity=commodity,

        state=state,

        district=None,

        market=(
            f"{state} market aggregate"
        ),

        variety=None,

        grade=None,

        arrival_date=arrival_date,

        min_price=(
            min(min_prices)
            if min_prices
            else None
        ),

        max_price=(
            max(max_prices)
            if max_prices
            else None
        ),

        modal_price=float(
            median(
                modal_prices
            )
        ),

        data_mode="LIVE",

        match_level="STATE",
    )

# MAIN SERVICE

def fetch_latest_mandi_price(
    *,
    crop: str,
    location: str,
) -> MandiPriceResult:

    api_key = (
        os.getenv(
            "DATA_GOV_API_KEY",
            "",
        )
        .strip()
    )

    enabled = _env_flag(
        "MANDI_API_ENABLED",
        "true",
    )

    if not enabled:
        raise MandiPriceError(
            "Government mandi-price "
            "integration is disabled."
        )

    if not api_key:
        raise MandiPriceError(
            "DATA_GOV_API_KEY is "
            "not configured."
        )

    placeholder_keys = {
        "YOUR_DATA_GOV_IN_API_KEY",
        "YOUR_ACTUAL_DATA_GOV_API_KEY",
        "YOUR_REAL_API_KEY",
        "YOUR_GENERATED_KEY",
    }

    if (
        api_key.upper()
        in placeholder_keys
    ):
        raise MandiPriceError(
            "DATA_GOV_API_KEY still "
            "contains a placeholder "
            "value."
        )

    commodity = normalize_commodity(
        crop
    )

    (
        normalized_location,
        state_hint,
    ) = split_location_state(
        location
    )

    # -----------------------------------------------------
    # Cached fallback helper
    # -----------------------------------------------------

    def cached_fallback():
        return _load_cached_result(
            commodity=commodity,
            location=(
                normalized_location
            ),
            state=state_hint,
        )

    # -----------------------------------------------------
    # 1. EXACT DISTRICT
    # -----------------------------------------------------

    try:
        records = request_records(
            api_key=api_key,
            commodity=commodity,
            location=(
                normalized_location
            ),
            location_field=(
                "district"
            ),
            limit=50,
        )

    except (
        MandiServiceUnavailableError
    ):
        cached = cached_fallback()

        if cached is not None:
            return cached

        raise

    record = choose_latest_record(
        records,
        normalized_location,
    )

    if record is not None:
        result = _result_from_record(
            record=record,
            commodity=commodity,
            match_level="DISTRICT",
        )

        _save_cached_result(
            commodity=commodity,
            location=(
                normalized_location
            ),
            state=state_hint,
            result=result,
        )

        return result

    # -----------------------------------------------------
    # 2. EXACT / PARTIAL MARKET NAME
    # -----------------------------------------------------

    try:
        records = request_records(
            api_key=api_key,
            commodity=commodity,
            location=(
                normalized_location
            ),
            location_field=(
                "market"
            ),
            limit=50,
        )

    except (
        MandiServiceUnavailableError
    ):
        cached = cached_fallback()

        if cached is not None:
            return cached

        raise

    record = choose_latest_record(
        records,
        normalized_location,
    )

    if record is not None:
        result = _result_from_record(
            record=record,
            commodity=commodity,
            match_level="MARKET",
        )

        _save_cached_result(
            commodity=commodity,
            location=(
                normalized_location
            ),
            state=state_hint,
            result=result,
        )

        return result

    # -----------------------------------------------------
    # AUTO-RESOLVE LOCATION
    #
    # If farmer entered only "Nashik" or a mandi name,
    # resolve State + District from Government data.
    # -----------------------------------------------------

    mapped_district, mapped_state = (
        resolve_maharashtra_location(
            normalized_location
        )
    )

    resolved_district: str | None = (
        mapped_district
    )

    if not state_hint and mapped_state:
        state_hint = mapped_state

    if not state_hint:
        try:
            location_record = (
                _resolve_location_record(
                    api_key=api_key,
                    location=normalized_location,
                )
            )

        except MandiServiceUnavailableError:
            location_record = None

        if location_record is not None:
            state_hint = _optional_text(
                location_record.get(
                    "state"
                )
            )

            resolved_district = (
                _optional_text(
                    location_record.get(
                        "district"
                    )
                )
            )

    # -----------------------------------------------------
    # 3. STATE FALLBACK
    #
    # Runs only when the state is known.
    # This is not a broad nationwide query.
    # -----------------------------------------------------

    if state_hint:
        try:
            state_records = (
                request_records(
                    api_key=api_key,
                    commodity=(
                        commodity
                    ),
                    location=state_hint,
                    location_field=(
                        "state"
                    ),
                    limit=100,
                )
            )

        except (
            MandiServiceUnavailableError
        ):
            cached = (
                cached_fallback()
            )

            if cached is not None:
                return cached

            raise

        state_result = (
            _state_market_aggregate(
                records=(
                    state_records
                ),
                commodity=commodity,
                state=state_hint,
            )
        )

        if state_result is not None:
            if resolved_district:
                state_result.district = (
                    resolved_district
                )

            _save_cached_result(
                commodity=commodity,
                location=(
                    normalized_location
                ),
                state=state_hint,
                result=state_result,
            )

            return state_result

    # -----------------------------------------------------
    # 4. LAST SUCCESS CACHE
    # -----------------------------------------------------

    cached = cached_fallback()

    if cached is not None:
        return cached

    # -----------------------------------------------------
    # 5. NOTHING AVAILABLE
    # -----------------------------------------------------

    location_label = (
        normalized_location
    )

    if state_hint:
        location_label += (
            f", {state_hint}"
        )

    raise MandiPriceError(
        "No mandi price was found "
        f"for {commodity} in "
        f"{location_label}."
    )

def _clean_historical_prices(
    prices: list[float],
) -> list[float]:
    cleaned: list[float] = []

    for value in prices or []:
        try:
            price = float(value)

        except (
            TypeError,
            ValueError,
        ):
            continue

        if price > 0:
            cleaned.append(
                price
            )

    return cleaned


# =========================================================
# TREND
# =========================================================

def _calculate_trend(
    historical_prices: list[float],
    current_price: float,
    predicted_price: float,
) -> str:
    if len(
        historical_prices
    ) >= 2:
        old_price = (
            historical_prices[0]
        )

        latest_price = (
            historical_prices[-1]
        )

        if (
            latest_price
            > old_price * 1.02
        ):
            return "UP"

        if (
            latest_price
            < old_price * 0.98
        ):
            return "DOWN"

    difference = (
        predicted_price
        - current_price
    ) / max(
        current_price,
        1.0,
    )

    if difference > 0.02:
        return "UP"

    if difference < -0.02:
        return "DOWN"

    return "STABLE"


# =========================================================
# CONFIDENCE
# =========================================================

def _base_confidence(
    historical_prices: list[float],
) -> float:
    count = len(
        historical_prices
    )

    if count >= 10:
        return 92.0

    if count >= 6:
        return 87.0

    if count >= 3:
        return 80.0

    if count >= 1:
        return 72.0

    return 65.0


def _confidence(
    historical_prices: list[float],
    *,
    data_mode: str,
) -> float:
    confidence = (
        _base_confidence(
            historical_prices
        )
    )

    normalized_mode = (
        data_mode
        .strip()
        .upper()
    )

    # Fresh Government mandi data.
    if normalized_mode == "LIVE":
        confidence = max(
            confidence,
            78.0,
        )

    # Last successful Government data.
    elif normalized_mode == "CACHED":
        confidence = max(
            confidence,
            72.0,
        )

    return confidence


# =========================================================
# HISTORICAL PREDICTION
# =========================================================

def _historical_base_prediction(
    historical_prices: list[float],
) -> float:
    recent = (
        historical_prices[-5:]
    )

    if not recent:
        raise ValueError(
            "Historical prices are empty."
        )

    weighted_sum = 0.0
    total_weight = 0.0

    for index, price in enumerate(
        recent,
        start=1,
    ):
        weighted_sum += (
            price * index
        )

        total_weight += index

    weighted_average = (
        weighted_sum
        / total_weight
    )

    average_price = mean(
        recent
    )

    return (
        weighted_average * 0.75
        + average_price * 0.25
    )

# =========================================================
# MAIN PRICE PREDICTION SERVICE
# =========================================================

def calculate_price_prediction(
    *,
    crop: str,
    location: str,
    quality: str,
    quantity: float,
    season: str,
    historical_prices: list[float],
) -> dict[str, Any]:

    # -----------------------------------------------------
    # Validate / normalize input
    # -----------------------------------------------------

    crop_name = (
        crop.strip()
    )

    if not crop_name:
        raise ValueError(
            "Crop name is required."
        )

    location_name = (
        location.strip()
    )

    if not location_name:
        raise ValueError(
            "Location is required."
        )

    quality_name = (
        quality.strip()
        if quality
        else "Standard"
    )

    season_name = (
        season.strip()
        if season
        else "Normal"
    )

    if quantity <= 0:
        raise ValueError(
            "Quantity must be greater "
            "than zero."
        )

    clean_prices = (
        _clean_historical_prices(
            historical_prices
        )
    )

    crop_key = (
        crop_name.lower()
    )

    # -----------------------------------------------------
    # Mandi state
    # -----------------------------------------------------

    mandi_result: (
        MandiPriceResult | None
    ) = None

    mandi_current_price: (
        float | None
    ) = None

    mandi_data_mode: str | None = (
        None
    )

    mandi_price_used = False

    fallback_demo_used = False

    # -----------------------------------------------------
    # 1. GOVERNMENT MANDI PRICE
    # -----------------------------------------------------

    if _env_flag(
        "MANDI_API_ENABLED",
        "true",
    ):
        try:
            mandi_result = (
                fetch_latest_mandi_price(
                    crop=crop_name,
                    location=(
                        location_name
                    ),
                )
            )

            mandi_current_price = (
                float(
                    mandi_result.modal_price
                )
            )

            mandi_data_mode = str(
                getattr(
                    mandi_result,
                    "data_mode",
                    "LIVE",
                )
            ).upper()

            # Accepted Government-backed modes.
            if (
                mandi_data_mode
                not in {
                    "LIVE",
                    "CACHED",
                }
            ):
                mandi_data_mode = (
                    "LIVE"
                )

            mandi_price_used = True

        except MandiPriceError:
            mandi_result = None

            mandi_current_price = (
                None
            )

            mandi_data_mode = None

            mandi_price_used = False

    # -----------------------------------------------------
    # 2. SELECT CURRENT PRICE SOURCE
    #
    # Government mandi
    #       ↓
    # Historical
    #       ↓
    # DEMO
    # -----------------------------------------------------

    if (
        mandi_current_price
        is not None
    ):
        current_price = (
            mandi_current_price
        )

        data_mode = (
            mandi_data_mode
            or "LIVE"
        )

    elif clean_prices:
        current_price = (
            clean_prices[-1]
        )

        data_mode = (
            "HISTORICAL"
        )

    else:
        demo_enabled = (
            _env_flag(
                "DEMO_MODE",
                "true",
            )
        )

        if not demo_enabled:
            raise ValueError(
                "Unable to retrieve mandi "
                "price and no historical "
                "prices were supplied."
            )

        current_price = float(
            DEMO_PRICES.get(
                crop_key,
                2500.0,
            )
        )

        fallback_demo_used = True

        data_mode = "DEMO"

    # -----------------------------------------------------
    # 3. BASE PREDICTION
    # -----------------------------------------------------

    if clean_prices:
        historical_prediction = (
            _historical_base_prediction(
                clean_prices
            )
        )

        # Government market price +
        # farmer historical data.
        if mandi_price_used:
            base_prediction = (
                current_price * 0.60
                + historical_prediction
                * 0.40
            )

        else:
            base_prediction = (
                historical_prediction
            )

    else:
        # Small near-term movement when
        # no historical series exists.
        base_prediction = (
            current_price * 1.02
        )

    # -----------------------------------------------------
    # 4. QUALITY ADJUSTMENT
    # -----------------------------------------------------

    quality_multiplier = (
        QUALITY_MULTIPLIERS.get(
            quality_name.lower(),
            1.0,
        )
    )

    # -----------------------------------------------------
    # 5. SEASON ADJUSTMENT
    # -----------------------------------------------------

    season_multiplier = (
        SEASON_MULTIPLIERS.get(
            season_name.lower(),
            1.0,
        )
    )

    # -----------------------------------------------------
    # 6. QUANTITY ADJUSTMENT
    # -----------------------------------------------------

    if quantity >= 100:
        quantity_multiplier = (
            0.97
        )

    elif quantity >= 50:
        quantity_multiplier = (
            0.985
        )

    else:
        quantity_multiplier = (
            1.0
        )

    # -----------------------------------------------------
    # 7. FINAL PREDICTED PRICE
    # -----------------------------------------------------

    predicted_price = (
        base_prediction
        * quality_multiplier
        * season_multiplier
        * quantity_multiplier
    )

    # Protect heuristic model against
    # unrealistic extreme predictions.
    min_allowed = (
        current_price * 0.75
    )

    max_allowed = (
        current_price * 1.35
    )

    predicted_price = max(
        min_allowed,
        min(
            predicted_price,
            max_allowed,
        ),
    )

    # -----------------------------------------------------
    # 8. CONFIDENCE
    # -----------------------------------------------------

    confidence = _confidence(
        clean_prices,
        data_mode=data_mode,
    )

    # Prediction range depends on
    # confidence.
    if confidence >= 85:
        range_percentage = (
            0.06
        )

    elif confidence >= 75:
        range_percentage = (
            0.08
        )

    else:
        range_percentage = (
            0.10
        )

    expected_min = (
        predicted_price
        * (
            1
            - range_percentage
        )
    )

    expected_max = (
        predicted_price
        * (
            1
            + range_percentage
        )
    )

    # -----------------------------------------------------
    # 9. TREND
    # -----------------------------------------------------

    trend = _calculate_trend(
        clean_prices,
        current_price,
        predicted_price,
    )

    # -----------------------------------------------------
    # 10. EXPLAINABLE FACTORS
    # -----------------------------------------------------

    base_adjustment_pct = (
        (
            base_prediction
            / max(
                current_price,
                1.0,
            )
        )
        - 1
    ) * 100

    quality_adjustment_pct = (
        quality_multiplier - 1
    ) * 100

    season_adjustment_pct = (
        season_multiplier - 1
    ) * 100

    quantity_adjustment_pct = (
        quantity_multiplier - 1
    ) * 100

    price_change_percentage = (
        (
            predicted_price
            - current_price
        )
        / max(
            current_price,
            1.0,
        )
        * 100
    )

    prediction_factors = {
        "base_price": (
            _round_price(
                current_price
            )
        ),

        "base_adjustment_percent": (
            _round_percentage(
                base_adjustment_pct
            )
        ),

        "quality_adjustment_percent": (
            _round_percentage(
                quality_adjustment_pct
            )
        ),

        "season_adjustment_percent": (
            _round_percentage(
                season_adjustment_pct
            )
        ),

        "quantity_adjustment_percent": (
            _round_percentage(
                quantity_adjustment_pct
            )
        ),
    }

    # -----------------------------------------------------
    # 11. UNIT
    # -----------------------------------------------------

    unit = (
        mandi_result.unit
        if mandi_result
        else "INR/quintal"
    )

    # -----------------------------------------------------
    # 12. RESPONSE
    # -----------------------------------------------------

    return {
        "current_price": (
            _round_price(
                current_price
            )
        ),

        "predicted_price": (
            _round_price(
                predicted_price
            )
        ),

        "expected_range_min": (
            _round_price(
                expected_min
            )
        ),

        "expected_range_max": (
            _round_price(
                expected_max
            )
        ),

        "confidence": (
            float(confidence)
        ),

        "trend": trend,

        "historical_prices": (
            clean_prices
        ),

        # True only when AGRIVA demo
        # fallback prices were actually used.
        "demo_mode": (
            fallback_demo_used
        ),

        # LIVE
        # CACHED
        # HISTORICAL
        # DEMO
        "data_mode": data_mode,

        "unit": unit,

        "market_data": (
            _market_data(
                mandi_result
            )
        ),

        "prediction_factors": (
            prediction_factors
        ),

        "price_change_percentage": (
            _round_percentage(
                price_change_percentage
            )
        ),
    }
from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class NormalizedStoredPrediction:
    current_price_per_kg: float
    predicted_price_per_kg: float
    expected_min_price_per_kg: float | None
    expected_max_price_per_kg: float | None
    compatibility_mode: str


def _clean_unit(unit: str | None) -> str:
    if not unit:
        raise ValueError(
            "Price unit is missing; cannot safely convert to INR/kg."
        )

    normalized = (
        str(unit)
        .strip()
        .lower()
        .replace("₹", "inr")
        .replace("rs.", "inr")
        .replace("rs", "inr")
        .replace(" ", "")
        .replace("_", "")
    )
    return normalized


def price_to_per_kg(
    value: float | int,
    unit: str | None,
) -> float:
    price = float(value)
    if price <= 0:
        raise ValueError("Price must be greater than zero.")

    normalized = _clean_unit(unit)

    if "quintal" in normalized or "qtl" in normalized:
        return round(price / 100.0, 4)

    if "/kg" in normalized or "perkg" in normalized:
        return round(price, 4)

    raise ValueError(
        f"Unsupported price unit '{unit}'. "
        "Expected INR/quintal (or qtl) or INR/kg."
    )


def normalize_stored_prediction(
    *,
    current_price: float | None,
    predicted_price: float,
    expected_min_price: float | None,
    expected_max_price: float | None,
    listing_reference_price_per_kg: float,
) -> NormalizedStoredPrediction:
    reference = float(listing_reference_price_per_kg)
    if reference <= 0:
        raise ValueError(
            "Listing reference price must be greater than zero."
        )

    if current_price is None:
        raise ValueError(
            "Stored prediction has no current price. "
            "Create a fresh price prediction before using sale decision."
        )

    current = float(current_price)
    predicted = float(predicted_price)

    if current <= 0 or predicted <= 0:
        raise ValueError(
            "Stored price prediction contains a non-positive price."
        )

    raw_ratio = current / reference
    quintal_ratio = (current / 100.0) / reference

    def plausible(ratio: float) -> bool:
        return 0.20 <= ratio < 5.00

    raw_is_per_kg = plausible(raw_ratio)
    divided_is_per_kg = plausible(quintal_ratio)

    if raw_is_per_kg == divided_is_per_kg:
        raise ValueError(
            "Stored prediction unit cannot be determined safely. "
            "Create a fresh price prediction for this crop/listing."
        )

    factor = 1.0 if raw_is_per_kg else 0.01
    mode = (
        "stored_per_kg"
        if raw_is_per_kg
        else "legacy_inr_per_quintal_converted"
    )

    def converted(value: float | None) -> float | None:
        if value is None:
            return None
        numeric = float(value)
        if numeric <= 0:
            raise ValueError(
                "Stored price prediction contains a non-positive range value."
            )
        return round(numeric * factor, 4)

    return NormalizedStoredPrediction(
        current_price_per_kg=round(current * factor, 4),
        predicted_price_per_kg=round(predicted * factor, 4),
        expected_min_price_per_kg=converted(expected_min_price),
        expected_max_price_per_kg=converted(expected_max_price),
        compatibility_mode=mode,
    )
