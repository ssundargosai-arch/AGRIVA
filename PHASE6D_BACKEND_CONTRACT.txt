from datetime import datetime
from uuid import UUID

from pydantic import BaseModel, ConfigDict, Field


class BuyerPlatformHistory(BaseModel):
    orders_recorded: int = 0
    completed_orders_recorded: int = 0
    payments_recorded: int = 0
    paid_payment_records: int = 0
    open_grievances_recorded: int = 0


class BuyerTrustSnapshot(BaseModel):
    buyer_id: UUID
    business_name: str | None = None
    business_type: str | None = None
    city: str | None = None
    state: str | None = None
    verification_status: str
    verification_note: str
    platform_history: BuyerPlatformHistory
    history_note: str


class PaymentAssuranceSnapshot(BaseModel):
    payment_exists: bool
    payment_id: UUID | None = None
    amount: float | None = None
    payment_method: str | None = None
    status: str | None = None
    transaction_reference: str | None = None
    transaction_status: str | None = None
    evidence_level: str
    evidence_note: str


class TransportAssuranceSnapshot(BaseModel):
    booking_exists: bool
    booking_id: UUID | None = None
    vehicle_type: str | None = None
    pickup_location: str | None = None
    delivery_location: str | None = None
    estimated_cost: float | None = None
    status: str | None = None


class TrackingEventSnapshot(BaseModel):
    id: UUID
    status: str
    message: str | None = None
    created_at: datetime

    model_config = ConfigDict(from_attributes=True)


class TransactionSafetyFlags(BaseModel):
    buyer_identity_verified_by_agriva: bool = False
    bank_payment_verified: bool = False
    payment_guaranteed: bool = False
    delivery_guaranteed: bool = False
    mutates_order: bool = False
    mutates_payment: bool = False
    mutates_transport: bool = False


class GrievanceCreate(BaseModel):
    order_id: UUID
    category: str = Field(min_length=2, max_length=40)
    subject: str = Field(min_length=3, max_length=180)
    description: str = Field(min_length=10, max_length=3000)


class GrievanceAdminUpdate(BaseModel):
    status: str = Field(min_length=2, max_length=30)
    resolution_note: str | None = Field(
        default=None,
        max_length=3000,
    )


class GrievanceResponse(BaseModel):
    id: UUID
    order_id: UUID
    farmer_user_id: UUID
    category: str
    subject: str
    description: str
    status: str
    resolution_note: str | None = None
    resolved_by_user_id: UUID | None = None
    resolved_at: datetime | None = None
    created_at: datetime
    updated_at: datetime

    model_config = ConfigDict(from_attributes=True)


class TransactionAssuranceOrderResponse(BaseModel):
    order_id: UUID
    listing_id: UUID
    order_status: str
    order_created_at: datetime
    total_amount: float
    crop: str | None = None
    quantity_kg: float
    price_per_kg: float | None = None
    buyer: BuyerTrustSnapshot
    payment: PaymentAssuranceSnapshot
    transport: TransportAssuranceSnapshot
    tracking: list[TrackingEventSnapshot]
    grievances: list[GrievanceResponse]
    safety: TransactionSafetyFlags

def _farmer_profile(
    db: Session,
    current_user: User,
) -> FarmerProfile:
    profile = (
        db.query(FarmerProfile)
        .filter(FarmerProfile.user_id == current_user.id)
        .first()
    )

    if profile is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Farmer profile not found",
        )

    return profile


def _farmer_order_query(
    db: Session,
    current_user: User,
):
    farmer = _farmer_profile(db, current_user)

    return (
        db.query(Order)
        .join(
            CropListing,
            Order.listing_id == CropListing.id,
        )
        .join(
            Crop,
            CropListing.crop_id == Crop.id,
        )
        .join(
            FarmField,
            Crop.field_id == FarmField.id,
        )
        .join(
            Farm,
            FarmField.farm_id == Farm.id,
        )
        .filter(Farm.farmer_id == farmer.id)
    )


def _get_farmer_order(
    db: Session,
    current_user: User,
    order_id: UUID,
) -> Order:
    order = (
        _farmer_order_query(db, current_user)
        .filter(Order.id == order_id)
        .first()
    )

    if order is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Order not found",
        )

    return order

def _buyer_snapshot(
    db: Session,
    order: Order,
) -> BuyerTrustSnapshot:
    buyer = (
        db.query(Buyer)
        .filter(Buyer.id == order.buyer_id)
        .first()
    )

    profile = None
    if buyer is not None:
        profile = (
            db.query(BuyerProfile)
            .filter(BuyerProfile.id == buyer.buyer_profile_id)
            .first()
        )

    orders_recorded = (
        db.query(Order)
        .filter(Order.buyer_id == order.buyer_id)
        .count()
    )

    completed_orders_recorded = (
        db.query(Order)
        .filter(
            Order.buyer_id == order.buyer_id,
            func.lower(Order.status) == "completed",
        )
        .count()
    )

    payments_recorded = (
        db.query(Payment)
        .join(Order, Order.id == Payment.order_id)
        .filter(Order.buyer_id == order.buyer_id)
        .count()
    )

    paid_payment_records = (
        db.query(Payment)
        .join(Order, Order.id == Payment.order_id)
        .filter(
            Order.buyer_id == order.buyer_id,
            func.lower(Payment.status).in_(PAID_LIKE_STATUSES),
        )
        .count()
    )

    open_grievances_recorded = (
        db.query(Grievance)
        .join(Order, Order.id == Grievance.order_id)
        .filter(
            Order.buyer_id == order.buyer_id,
            Grievance.status.in_(OPEN_GRIEVANCE_STATUSES),
        )
        .count()
    )

    return BuyerTrustSnapshot(
        buyer_id=order.buyer_id,
        business_name=(
            profile.business_name if profile is not None else None
        ),
        business_type=(
            profile.business_type if profile is not None else None
        ),
        city=(profile.city if profile is not None else None),
        state=(profile.state if profile is not None else None),
        verification_status=BUYER_VERIFICATION_STATUS,
        verification_note=BUYER_VERIFICATION_NOTE,
        platform_history=BuyerPlatformHistory(
            orders_recorded=orders_recorded,
            completed_orders_recorded=completed_orders_recorded,
            payments_recorded=payments_recorded,
            paid_payment_records=paid_payment_records,
            open_grievances_recorded=open_grievances_recorded,
        ),
        history_note=BUYER_HISTORY_NOTE,
    )


def _payment_snapshot(
    db: Session,
    order: Order,
) -> PaymentAssuranceSnapshot:
    payment = (
        db.query(Payment)
        .filter(Payment.order_id == order.id)
        .first()
    )

    if payment is None:
        evidence_level, evidence_note = payment_evidence(
            payment_exists=False,
            transaction_reference=None,
        )

        return PaymentAssuranceSnapshot(
            payment_exists=False,
            evidence_level=evidence_level,
            evidence_note=evidence_note,
        )

    transaction = (
        db.query(Transaction)
        .filter(Transaction.payment_id == payment.id)
        .first()
    )

    reference = (
        transaction.transaction_reference
        if transaction is not None
        else None
    )

    evidence_level, evidence_note = payment_evidence(
        payment_exists=True,
        transaction_reference=reference,
    )

    return PaymentAssuranceSnapshot(
        payment_exists=True,
        payment_id=payment.id,
        amount=float(payment.amount),
        payment_method=payment.payment_method,
        status=payment.status,
        transaction_reference=reference,
        transaction_status=(
            transaction.status if transaction is not None else None
        ),
        evidence_level=evidence_level,
        evidence_note=evidence_note,
    )


def _transport_snapshot(
    db: Session,
    order: Order,
) -> TransportAssuranceSnapshot:
    booking = (
        db.query(TransportBooking)
        .filter(TransportBooking.order_id == order.id)
        .first()
    )

    if booking is None:
        return TransportAssuranceSnapshot(
            booking_exists=False,
        )

    return TransportAssuranceSnapshot(
        booking_exists=True,
        booking_id=booking.id,
        vehicle_type=booking.vehicle_type,
        pickup_location=booking.pickup_location,
        delivery_location=booking.delivery_location,
        estimated_cost=float(booking.estimated_cost),
        status=booking.status,
    )


def _tracking_snapshots(
    db: Session,
    order: Order,
) -> list[TrackingEventSnapshot]:
    events = (
        db.query(OrderTrackingEvent)
        .filter(OrderTrackingEvent.order_id == order.id)
        .order_by(OrderTrackingEvent.created_at.asc())
        .all()
    )

    return [
        TrackingEventSnapshot.model_validate(event)
        for event in events
    ]


def _grievances_for_order(
    db: Session,
    order: Order,
) -> list[GrievanceResponse]:
    rows = (
        db.query(Grievance)
        .filter(Grievance.order_id == order.id)
        .order_by(Grievance.created_at.desc())
        .all()
    )

    return [
        GrievanceResponse.model_validate(row)
        for row in rows
    ]


def _order_crop_and_quantity(
    db: Session,
    order: Order,
) -> tuple[str | None, float, float | None]:
    items = (
        db.query(OrderItem)
        .filter(OrderItem.order_id == order.id)
        .all()
    )

    quantity = sum(
        float(item.quantity_kg)
        for item in items
    )

    price_per_kg = None
    if items:
        weighted_total = sum(
            float(item.quantity_kg) * float(item.price_per_kg)
            for item in items
        )
        if quantity > 0:
            price_per_kg = weighted_total / quantity

    listing = (
        db.query(CropListing)
        .filter(CropListing.id == order.listing_id)
        .first()
    )

    crop_name = None
    if listing is not None:
        crop = (
            db.query(Crop)
            .filter(Crop.id == listing.crop_id)
            .first()
        )
        if crop is not None:
            crop_name = crop.name

    return crop_name, quantity, price_per_kg


def _build_order_assurance(
    db: Session,
    order: Order,
) -> TransactionAssuranceOrderResponse:
    crop_name, quantity, price_per_kg = _order_crop_and_quantity(
        db,
        order,
    )

    return TransactionAssuranceOrderResponse(
        order_id=order.id,
        listing_id=order.listing_id,
        order_status=order.status,
        order_created_at=order.created_at,
        total_amount=float(order.total_amount),
        crop=crop_name,
        quantity_kg=quantity,
        price_per_kg=price_per_kg,
        buyer=_buyer_snapshot(db, order),
        payment=_payment_snapshot(db, order),
        transport=_transport_snapshot(db, order),
        tracking=_tracking_snapshots(db, order),
        grievances=_grievances_for_order(db, order),
        safety=TransactionSafetyFlags(),
    )

@router.post(
    "/grievances",
    response_model=GrievanceResponse,
    status_code=status.HTTP_201_CREATED,
)
def create_grievance(
    payload: GrievanceCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(require_farmer),
):
    order = _get_farmer_order(
        db,
        current_user,
        payload.order_id,
    )

    try:
        category = normalize_grievance_category(payload.category)
    except AssuranceValidationError as exc:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=str(exc),
        ) from exc

    existing = (
        db.query(Grievance)
        .filter(
            Grievance.order_id == order.id,
            Grievance.farmer_user_id == current_user.id,
            Grievance.category == category,
            Grievance.status.in_(OPEN_GRIEVANCE_STATUSES),
        )
        .first()
    )

    if existing is not None:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail=(
                "An open grievance already exists for this order and category."
            ),
        )

    subject = payload.subject.strip()
    description = payload.description.strip()

    grievance = Grievance(
        order_id=order.id,
        farmer_user_id=current_user.id,
        category=category,
        subject=subject,
        description=description,
        status="OPEN",
    )

    try:
        db.add(grievance)
        db.commit()
        db.refresh(grievance)
    except Exception as exc:
        db.rollback()
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Unable to create grievance",
        ) from exc

    return grievance

@router.get(
    "/admin/grievances",
    response_model=list[GrievanceResponse],
)
def admin_grievances(
    grievance_status: str | None = Query(
        default=None,
        alias="status",
    ),
    category: str | None = Query(default=None),
    db: Session = Depends(get_db),
    current_user: User = Depends(require_admin),
):
    query = db.query(Grievance)

    if grievance_status:
        try:
            normalized_status = normalize_grievance_status(
                grievance_status
            )
        except AssuranceValidationError as exc:
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
                detail=str(exc),
            ) from exc

        query = query.filter(
            Grievance.status == normalized_status
        )

    if category:
        try:
            normalized_category = normalize_grievance_category(category)
        except AssuranceValidationError as exc:
            raise HTTPException(
                status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
                detail=str(exc),
            ) from exc

        query = query.filter(
            Grievance.category == normalized_category
        )

    return (
        query
        .order_by(Grievance.created_at.desc())
        .limit(300)
        .all()
    )


@router.patch(
    "/admin/grievances/{grievance_id}",
    response_model=GrievanceResponse,
)
def admin_update_grievance(
    grievance_id: UUID,
    payload: GrievanceAdminUpdate,
    db: Session = Depends(get_db),
    current_user: User = Depends(require_admin),
):
    grievance = (
        db.query(Grievance)
        .filter(Grievance.id == grievance_id)
        .first()
    )

    if grievance is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Grievance not found",
        )

    resolution_note = (
        payload.resolution_note.strip()
        if payload.resolution_note
        else None
    )

    try:
        new_status = validate_grievance_transition(
            grievance.status,
            payload.status,
            resolution_note,
        )
    except AssuranceValidationError as exc:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=str(exc),
        ) from exc

    grievance.status = new_status

    if resolution_note:
        grievance.resolution_note = resolution_note

    if new_status in {"RESOLVED", "REJECTED"}:
        grievance.resolved_by_user_id = current_user.id
        grievance.resolved_at = datetime.now(timezone.utc)
    else:
        grievance.resolved_by_user_id = None
        grievance.resolved_at = None

    try:
        db.commit()
        db.refresh(grievance)
    except Exception as exc:
        db.rollback()
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Unable to update grievance",
        ) from exc

    return grievance

ALLOWED_GRIEVANCE_CATEGORIES = {
    "PAYMENT",
    "DELIVERY",
    "QUALITY",
    "BUYER_COMMUNICATION",
    "OTHER",
}

ALLOWED_GRIEVANCE_STATUSES = {
    "OPEN",
    "IN_REVIEW",
    "RESOLVED",
    "REJECTED",
}

TERMINAL_GRIEVANCE_STATUSES = {
    "RESOLVED",
    "REJECTED",
}

BUYER_VERIFICATION_STATUS = "NOT_VERIFIED_BY_AGRIVA"
BUYER_VERIFICATION_NOTE = (
    "AGRIVA has profile and platform-activity records for this buyer, "
    "but Phase 6D does not claim government, KYC, bank, or independent "
    "identity verification."
)

BUYER_HISTORY_NOTE = (
    "Platform history is based only on AGRIVA database records. "
    "It is not a credit score, identity verification, or guarantee "
    "of future payment or delivery."
)

PAYMENT_REFERENCE_NOTE = (
    "The AGRIVA transaction reference is an internal platform record. "
    "It is not independent bank confirmation and does not guarantee "
    "settlement."
)

PAYMENT_RECORD_NOTE = (
    "A payment record exists in AGRIVA, but no transaction reference "
    "is recorded. This is not bank verification."
)

NO_PAYMENT_NOTE = (
    "No payment record is stored for this order in AGRIVA."
)


class AssuranceValidationError(ValueError):
    pass


def normalize_grievance_category(value: str) -> str:
    category = value.strip().upper()

    if category not in ALLOWED_GRIEVANCE_CATEGORIES:
        raise AssuranceValidationError(
            "Invalid grievance category. Allowed: "
            + ", ".join(sorted(ALLOWED_GRIEVANCE_CATEGORIES))
        )

    return category


def normalize_grievance_status(value: str) -> str:
    grievance_status = value.strip().upper()

    if grievance_status not in ALLOWED_GRIEVANCE_STATUSES:
        raise AssuranceValidationError(
            "Invalid grievance status. Allowed: "
            + ", ".join(sorted(ALLOWED_GRIEVANCE_STATUSES))
        )

    return grievance_status


def validate_grievance_transition(
    current_status: str,
    requested_status: str,
    resolution_note: str | None,
) -> str:
    current = normalize_grievance_status(current_status)
    requested = normalize_grievance_status(requested_status)

    if requested == current:
        if requested in TERMINAL_GRIEVANCE_STATUSES:
            if not (resolution_note or "").strip():
                raise AssuranceValidationError(
                    "Resolution note is required for terminal grievance status."
                )
        return requested

    transitions = {
        "OPEN": {"IN_REVIEW", "RESOLVED", "REJECTED"},
        "IN_REVIEW": {"RESOLVED", "REJECTED"},
        "RESOLVED": set(),
        "REJECTED": set(),
    }

    if requested not in transitions[current]:
        raise AssuranceValidationError(
            f"Invalid grievance status transition: {current} -> {requested}"
        )

    if requested in TERMINAL_GRIEVANCE_STATUSES:
        if not (resolution_note or "").strip():
            raise AssuranceValidationError(
                "Resolution note is required when resolving or rejecting a grievance."
            )

    return requested


def payment_evidence(
    *,
    payment_exists: bool,
    transaction_reference: str | None,
) -> tuple[str, str]:
    if not payment_exists:
        return "NO_PAYMENT_RECORD", NO_PAYMENT_NOTE

    if transaction_reference:
        return (
            "AGRIVA_TRANSACTION_REFERENCE_RECORDED",
            PAYMENT_REFERENCE_NOTE,
        )

    return "PAYMENT_RECORD_ONLY", PAYMENT_RECORD_NOTE

from datetime import datetime

from sqlalchemy import (
    CheckConstraint,
    DateTime,
    ForeignKey,
    String,
    Text,
    func,
)
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import Mapped, mapped_column

from app.db import Base


class Grievance(Base):
    """
    Farmer-raised transaction grievance.

    This record is an AGRIVA support/audit record only. It does not:
    - prove that a buyer identity is verified,
    - prove that a bank payment succeeded,
    - change an order, payment, or transport booking automatically,
    - guarantee resolution or compensation.
    """

    __tablename__ = "transaction_grievances"
    __table_args__ = (
        CheckConstraint(
            "category IN ('PAYMENT', 'DELIVERY', 'QUALITY', "
            "'BUYER_COMMUNICATION', 'OTHER')",
            name="ck_transaction_grievance_category",
        ),
        CheckConstraint(
            "status IN ('OPEN', 'IN_REVIEW', 'RESOLVED', 'REJECTED')",
            name="ck_transaction_grievance_status",
        ),
    )

    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=uuid.uuid4,
    )

    order_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("orders.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    farmer_user_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("users.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    category: Mapped[str] = mapped_column(
        String(40),
        nullable=False,
        index=True,
    )

    subject: Mapped[str] = mapped_column(
        String(180),
        nullable=False,
    )

    description: Mapped[str] = mapped_column(
        Text,
        nullable=False,
    )

    status: Mapped[str] = mapped_column(
        String(30),
        default="OPEN",
        nullable=False,
        index=True,
    )

    resolution_note: Mapped[str | None] = mapped_column(
        Text,
        nullable=True,
    )

    resolved_by_user_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("users.id", ondelete="SET NULL"),
        nullable=True,
        index=True,
    )

    resolved_at: Mapped[datetime | None] = mapped_column(
        DateTime(timezone=True),
        nullable=True,
    )

    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        server_default=func.now(),
        nullable=False,
    )

    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        server_default=func.now(),
        onupdate=func.now(),
        nullable=False,
    )

import pytest

from app.api.routes.transaction_assurance import (
    transaction_assurance_health,
)
from app.services.transaction_assurance import (
    AssuranceValidationError,
    normalize_grievance_category,
    normalize_grievance_status,
    payment_evidence,
    validate_grievance_transition,
)


def test_phase6d_health_safety_flags():
    result = transaction_assurance_health()

    assert result["status"] == "ok"
    assert result["phase"] == "SIH-6D"
    assert result["bank_payment_verification"] is False
    assert result["payment_guarantee"] is False
    assert result["delivery_guarantee"] is False
    assert result["grievance_mutates_order"] is False
    assert result["grievance_mutates_payment"] is False
    assert result["grievance_mutates_transport"] is False


def test_grievance_category_normalization():
    assert normalize_grievance_category(" payment ") == "PAYMENT"
    assert normalize_grievance_category("buyer_communication") == (
        "BUYER_COMMUNICATION"
    )


def test_invalid_grievance_category_fails_closed():
    with pytest.raises(AssuranceValidationError):
        normalize_grievance_category("bank_verified")


def test_grievance_status_normalization():
    assert normalize_grievance_status(" in_review ") == "IN_REVIEW"


def test_open_to_in_review_transition():
    assert validate_grievance_transition(
        "OPEN",
        "IN_REVIEW",
        None,
    ) == "IN_REVIEW"


def test_terminal_transition_requires_resolution_note():
    with pytest.raises(AssuranceValidationError):
        validate_grievance_transition(
            "IN_REVIEW",
            "RESOLVED",
            None,
        )


def test_terminal_grievance_cannot_reopen():
    with pytest.raises(AssuranceValidationError):
        validate_grievance_transition(
            "RESOLVED",
            "OPEN",
            "Already resolved",
        )


def test_payment_evidence_no_record():
    level, note = payment_evidence(
        payment_exists=False,
        transaction_reference=None,
    )

    assert level == "NO_PAYMENT_RECORD"
    assert "No payment record" in note


def test_payment_evidence_record_only():
    level, note = payment_evidence(
        payment_exists=True,
        transaction_reference=None,
    )

    assert level == "PAYMENT_RECORD_ONLY"
    assert "not bank verification" in note


def test_transaction_reference_is_not_bank_verification():
    level, note = payment_evidence(
        payment_exists=True,
        transaction_reference="AGRIVA-ABC123",
    )

    assert level == "AGRIVA_TRANSACTION_REFERENCE_RECORDED"
    assert "not independent bank confirmation" in note
